### Input:

- La interface `{ts icon} ControlValueAccesor` indica que el componente personalizado que estamos creando ahora provee valores validos para los formularios de Angular.

```ts title:InputComponent
@Component({
  selector: 'app-input',
  standalone: true,
  imports: [],
  providers: [
    { provide: NG_VALUE_ACCESSOR, useExisting: forwardRef(() => InputComponent), multi: true }
  ],
  templateUrl: './input.component.html',
  styleUrl: './input.component.scss'
})
export class InputComponent implements ControlValueAccessor {

  @Input({ required: true }) type!: 'text';

  value: any = '';
  disabled = false;

  private onChange: (value: any) => void = () => {};
  private onTouched: () => void = () => {};


  //? For reactive Forms:
  writeValue(value: any): void {
    this.value = value ?? '';
  }

  registerOnChange(fn: any): void {
    this.onChange = fn;
  }

  registerOnTouched(fn: any): void {
    this.onTouched = fn;
  }

  setDisabledState(isDisabled: boolean): void {
    this.disabled = isDisabled;
  }

  onBlur(): void {
    this.onTouched();
  }



  //? All:
  onInput(e: Event): void {
    this.value = (e.target as HTMLInputElement).value;
    this.onChange(this.value);
  }

}
```

### Uso:
```html
<app-input [(ngModel)]="nombre" />

<app-input formControlName="nombre" />
```

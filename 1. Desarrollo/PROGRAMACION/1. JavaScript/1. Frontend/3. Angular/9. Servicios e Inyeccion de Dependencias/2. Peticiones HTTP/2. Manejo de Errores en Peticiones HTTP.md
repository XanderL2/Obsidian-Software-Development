## Conceptos Previos:
- Un observable cuando salta un error se termina, al igual que con el `{ts icon}complete()`.
- `{ts icon} catchError()` es un Pipe que permite manejar el error, **si o si debe devolver un observable valido.**
- `{ts icon}throwError()` es un Pipe que sirve para lanzar un error en un observable.



### Peticiones HTTP:

- **Servicio:**
	```ts
	@Injectable({
	  providedIn: 'root',
	})
	export class PostsService {
	
	  private BASE_ENDPOINT = "https://jsonplaceholder.typicode.com/posts/";
	  private http = inject(HttpClient);
	
	
	  public getPosts(): Observable<Post[]> {
	  
	    return this.http.get<Post[]>(this.BASE_ENDPOINT).pipe(
	      catchError((e: HttpErrorResponse) => { // 👈 Manejamos el error
	        return throwError(() => e.message) // 👈 Obtiene el message del error de la api y lo lanza de nuevo
	      })
	    );
	  }
	}
	```

- **Component:**
	
	```ts
	@Component({
	  selector: 'app-root',
	  templateUrl: './app.component.html',
	  imports: [CommonModule]
	})
	export class AppComponent {
	
	  private postsService = inject(PostsService);
	  errorMessage: string = "";
	
	  $posts: Observable<Post[] | null> = this.postsService.getPosts().pipe(
	    catchError((e) => { 
	      this.errorMessage = e; // 👈 Obtenemos el message lanzado por el servicio
	      return of<Post[]>([]); // 👈 Emitimos un observable vacio para que no muera el observable
	    })
	  );
	
	}
	```

- **Component HTML:**
	```html
	@let posts = ($posts | async); 
	
	
	@if (errorMessage) {
	  <p>{{errorMessage}}</p>
	
	} @else if (!posts) {
	  <p>loading</p>  
	
	} @else {
	
	  @for (post of posts; track post.id) {
	    <p>{{post.title}}</p>
	
	  }@empty {
	    <p>Empty posts</p>
	  }
	
	}
	```

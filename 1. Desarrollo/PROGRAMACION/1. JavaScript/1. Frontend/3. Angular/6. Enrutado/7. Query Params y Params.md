## Params:

- **Definicion:**
	```ts
	const routes: Routes = [
	  { path: 'products/:id', component: ProductDetailComponent },
	];
	```
	
- **Seteo de params**
	- En el `{ts icon} Typescript`
		```ts
		this.router.navigate(['/products', id]);
		```

	- En el `{html icon}HTML`
		```html
		<a [routerLink]="['/products', id]">Ver Producto</a>	
		```


- **Obtencion**
	```ts
	import { ActivatedRoute } from '@angular/router';
	
	constructor(private route: ActivatedRoute) {}
	
	ngOnInit() {
	
	  // Si solo queremos obtener el value en concreto sin reaccionar
	  const id = this.route.snapshot.paramMap.get('id'); 

	  // Si queremos reaccionar a cambios de ellos	  
	  this.route.paramMap.subscribe(params => {
	     const id = params.get('id');
	   });
	}
	```


## Query Params:
 - **Seteo de query params:**
	-  En el `{ts icon} Typescript`:
		```ts
		this.router.navigate(['/products'], { queryParams: { page: 1, sort: 'asc' } });
		```
	 - En el `{html icon} HTML`:
		```html
		<a [routerLink]="['/products']" [queryParams]="{ page: 1, sort: 'asc' }">Ver Productos</a> 
		```
- **Obtencion**
```ts
import { ActivatedRoute } from '@angular/router';

constructor(private route: ActivatedRoute) {}

ngOnInit() {
  // Si solo queremos obtener el value en concreto sin reaccionar
  const page = this.route.snapshot.queryParamMap.get('page');

  // Si queremos reaccionar a cambios de ellos
  this.route.queryParamMap.subscribe(params => {
    const page = params.get('page');
    const sort = params.get('sort');
  });
}
```
